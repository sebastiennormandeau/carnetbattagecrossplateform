rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    function isAdmin() {
      return request.auth != null &&
             firestore.exists(/databases/(default)/documents/admins/$(request.auth.uid)) &&
             firestore.get(/databases/(default)/documents/admins/$(request.auth.uid)).data.enabled == true;
    }

    function hasProjectReadAccess(projectId) {
       let projectDoc = firestore.get(/databases/(default)/documents/projects/$(projectId));
       return request.auth != null && 
              (
                 projectDoc.data.ownerUid == request.auth.uid || 
                 (projectDoc.data.readUsers != null && request.auth.uid in projectDoc.data.readUsers)
              );
    }
    
    function hasProjectWriteAccess(projectId) {
       let projectDoc = firestore.get(/databases/(default)/documents/projects/$(projectId));
       return request.auth != null && 
              (
                 projectDoc.data.ownerUid == request.auth.uid || 
                 (projectDoc.data.writeUsers != null && request.auth.uid in projectDoc.data.writeUsers)
              );
    }

    match /projects/{projectId}/{allPaths=**} {
      allow read: if request.auth != null && (isAdmin() || hasProjectReadAccess(projectId));
      allow write: if isAdmin() || hasProjectWriteAccess(projectId);
    }

    // --- CHEMINS LÉGACY (NE PAS SUPPRIMER) ---
    match /plans/{ownerUid}/{allPaths=**} {
      allow read: if request.auth != null;
      allow write: if isAdmin() || request.auth.uid == ownerUid;
    }

    match /project_photos/{ownerUid}/{allPaths=**} {
      allow read, write: if isAdmin() || request.auth.uid == ownerUid;
    }
  }
}
