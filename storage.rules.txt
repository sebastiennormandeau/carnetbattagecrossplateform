rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {

    // Vérifie si l'utilisateur est admin via les Custom Claims
    function isAdmin() {
      return request.auth != null && request.auth.token.role == 'admin';
    }

    // Le compte de support central (Super-Admin)
    function isSuperAdmin() {
      return request.auth != null && firestore.get(/databases/(default)/documents/users/$(request.auth.uid)).data.isSuperAdmin == true;
    }

    // Vérifie si le projet parent appartient au même tenant que l'utilisateur actif (ou Super-Admin)
    function parentProjectIsSameTenant(projectId) {
       let projectDoc = firestore.get(/databases/(default)/documents/projects/$(projectId));
       return request.auth != null && (projectDoc.data.companyId == request.auth.token.companyId || isSuperAdmin());
    }

    // Vérifie si l'utilisateur a l'outil Carnet activé
    function hasCarnetAccess() {
      let userData = firestore.get(/databases/(default)/documents/users/$(request.auth.uid)).data;
      return !("tools" in userData) || !("carnet" in userData.tools) || userData.tools.carnet == true;
    }

    // Vérifie les droits de lecture (propriétaire, ou assignedUsers + Carnet)
    function hasProjectReadAccess(projectId) {
       let projectDoc = firestore.get(/databases/(default)/documents/projects/$(projectId));
       return request.auth != null && 
              (
                 projectDoc.data.ownerUid == request.auth.uid || 
                 (projectDoc.data.assignedUsers != null && request.auth.uid in projectDoc.data.assignedUsers && hasCarnetAccess())
              );
    }
    
    // Vérifie les droits d'écriture (propriétaire ou assignedUsers + Carnet)
    function hasProjectWriteAccess(projectId) {
       let projectDoc = firestore.get(/databases/(default)/documents/projects/$(projectId));
       return request.auth != null && 
              (
                 projectDoc.data.ownerUid == request.auth.uid || 
                 (projectDoc.data.assignedUsers != null && request.auth.uid in projectDoc.data.assignedUsers && hasCarnetAccess())
              );
    }

    // Uniquement la nouvelle architecture de stockage par projet
    match /projects/{projectId}/{allPaths=**} {
      allow read: if request.auth != null && parentProjectIsSameTenant(projectId) && (
                      isSuperAdmin() || isAdmin() || hasProjectReadAccess(projectId)
                  );
      allow write: if request.auth != null && parentProjectIsSameTenant(projectId) && (
                      isSuperAdmin() || isAdmin() || hasProjectWriteAccess(projectId)
                  );
    }

    // Support pour les anciens fichiers (Legacy Kotlin App)
    match /project_photos/{allPaths=**} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    match /project_docs/{allPaths=**} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    match /project_plans/{allPaths=**} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    match /plans/{allPaths=**} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

  }
}
