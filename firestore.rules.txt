rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Vérification de base : utilisateur authentifié avec un tenant actif (companyId)
    function hasTenant() {
      return request.auth != null && request.auth.token.companyId != null;
    }

    // Le compte de support central (Super-Admin)
    function isSuperAdmin() {
      return request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isSuperAdmin == true;
    }

    // L'utilisateur est admin de son tenant, ou Super-Admin
    function isAdmin() {
      return (request.auth != null && request.auth.token.role == 'admin') || isSuperAdmin();
    }

    // Le document appartient au même tenant que l'utilisateur (ou Super-Admin)
    function isSameTenant(docData) {
      return (request.auth.token.companyId != null && docData.companyId == request.auth.token.companyId) || isSuperAdmin();
    }

    // L'utilisateur est-il assigné à ce projet spécifique
    function isAssignedToProject(projectId) {
      return request.auth != null && 
             request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.assignedUsers;
    }

    // Vérifie que le projet parent appartient bien à la bonne compagnie (ou Super-Admin)
    function parentProjectIsSameTenant(projectId) {
      return (request.auth.token.companyId != null && get(/databases/$(database)/documents/projects/$(projectId)).data.companyId == request.auth.token.companyId) || isSuperAdmin();
    }

    // Vérifie si l'utilisateur a l'outil Carnet activé
    function hasCarnetAccess() {
      let userData = get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
      return !("tools" in userData) || !("carnet" in userData.tools) || userData.tools.carnet == true;
    }

    match /users/{userId} {
      // Un utilisateur peut TOUJOURS lire ses propres infos, ou celles des autres de son entreprise
      allow read: if request.auth.uid == userId || (hasTenant() && isSameTenant(resource.data)) || isSuperAdmin(); 
      // Un utilisateur peut se modifier lui-même, ou un admin de SON entreprise peut le modifier
      // Dans les deux cas, le nouveau document DOIT rester dans la même entreprise
      allow write: if request.auth.uid == userId || (hasTenant() && 
                   isSameTenant(request.resource.data) &&
                   (
                     (resource == null) || 
                     isSameTenant(resource.data)
                   ) &&
                   isAdmin()) || isSuperAdmin();
    }

    // Collection obsolète ?


    match /projects/{projectId} {
      allow read: if hasTenant() && isSameTenant(resource.data) && (
                      isAdmin() || 
                      resource.data.ownerUid == request.auth.uid ||
                      (request.auth.uid in resource.data.assignedUsers)
                  );
                      
      allow update: if hasTenant() && isSameTenant(resource.data) && isSameTenant(request.resource.data) && (
                        isAdmin() || 
                        resource.data.ownerUid == request.auth.uid ||
                        (request.auth.uid in resource.data.assignedUsers)
                    );
      
      allow create: if hasTenant() && isSameTenant(request.resource.data);
      allow delete: if hasTenant() && isSameTenant(resource.data) && isAdmin();

      // SOUS-COLLECTIONS de Projects
      // On vérifie que le projet parent appartient à l'entreprise ET que l'utilisateur a l'accès Carnet (ou Admin)
      match /piles/{pileId} {
        allow read: if hasTenant() && parentProjectIsSameTenant(projectId) && (isAdmin() || (hasCarnetAccess() && isAssignedToProject(projectId)));
        allow write: if hasTenant() && parentProjectIsSameTenant(projectId) && (isAdmin() || (hasCarnetAccess() && isAssignedToProject(projectId)));
      }
      match /photos/{photoId} {
        allow read: if hasTenant() && parentProjectIsSameTenant(projectId) && (isAdmin() || (hasCarnetAccess() && isAssignedToProject(projectId)));
        allow write: if hasTenant() && parentProjectIsSameTenant(projectId) && (isAdmin() || (hasCarnetAccess() && isAssignedToProject(projectId)));
      }
      match /notes/{noteId} {
        allow read: if hasTenant() && parentProjectIsSameTenant(projectId) && (isAdmin() || (hasCarnetAccess() && isAssignedToProject(projectId)));
        allow write: if hasTenant() && parentProjectIsSameTenant(projectId) && (isAdmin() || (hasCarnetAccess() && isAssignedToProject(projectId)));
      }
      match /docs/{docId} {
        allow read: if hasTenant() && parentProjectIsSameTenant(projectId) && (isAdmin() || (hasCarnetAccess() && isAssignedToProject(projectId)));
        allow write: if hasTenant() && parentProjectIsSameTenant(projectId) && (isAdmin() || (hasCarnetAccess() && isAssignedToProject(projectId)));
      }
      match /hotspots/{hotspotId} {
        allow read: if hasTenant() && parentProjectIsSameTenant(projectId) && (isAdmin() || (hasCarnetAccess() && isAssignedToProject(projectId)));
        allow write: if hasTenant() && parentProjectIsSameTenant(projectId) && (isAdmin() || (hasCarnetAccess() && isAssignedToProject(projectId)));
      }
    }

    match /calendar_events/{eventId} {
      allow read: if hasTenant() && isSameTenant(resource.data) && (
                     isAdmin() || (request.auth.uid in resource.data.assignedUsers)
                  );
      allow create: if hasTenant() && isSameTenant(request.resource.data) && isAdmin();
      allow update: if hasTenant() && isSameTenant(resource.data) && isSameTenant(request.resource.data) && isAdmin();
      allow delete: if hasTenant() && isSameTenant(resource.data) && isAdmin();
    }

    match /map_points/{pointId} {
      allow read: if hasTenant() && isSameTenant(resource.data);
      allow create: if hasTenant() && isSameTenant(request.resource.data);
      allow update, delete: if hasTenant() && isSameTenant(resource.data) && isSameTenant(request.resource.data) && isAdmin();
    }

    match /shifts/{shiftId} {
      allow read: if hasTenant() && isSameTenant(resource.data) && (isAdmin() || resource.data.userId == request.auth.uid);
      allow create: if hasTenant() && isSameTenant(request.resource.data) && request.resource.data.userId == request.auth.uid;
      allow update: if hasTenant() && isSameTenant(resource.data) && isSameTenant(request.resource.data) && resource.data.userId == request.auth.uid;
      allow delete: if hasTenant() && isSameTenant(resource.data) && isAdmin();
    }

    match /timesheet_entries/{entryId} {
      allow read: if hasTenant() && isSameTenant(resource.data) && (isAdmin() || resource.data.employeeId == request.auth.uid);
      allow create: if hasTenant() && isSameTenant(request.resource.data) && request.resource.data.employeeId == request.auth.uid;
      allow update: if hasTenant() && isSameTenant(resource.data) && isSameTenant(request.resource.data) && (isAdmin() || resource.data.employeeId == request.auth.uid);
      allow delete: if hasTenant() && isSameTenant(resource.data) && isAdmin();
    }

    match /settings/{docId} {
      allow read: if hasTenant() && (resource == null || resource.data.companyId == request.auth.token.companyId);
      allow create: if hasTenant() && request.resource.data.companyId == request.auth.token.companyId && isAdmin();
      allow update, delete: if hasTenant() && resource.data.companyId == request.auth.token.companyId && request.resource.data.companyId == request.auth.token.companyId && isAdmin();
    }

    match /hammers/{hammerId} {
      allow read: if hasTenant() && isSameTenant(resource.data);
      allow create: if hasTenant() && isSameTenant(request.resource.data) && 
                            request.resource.data.keys().hasAll(['capThicknessIn', 'capAreaSqIn', 'capModulusMPa']) &&
                            request.resource.data.capThicknessIn is number &&
                            request.resource.data.capAreaSqIn is number &&
                            request.resource.data.capModulusMPa is number;
      allow update: if hasTenant() && isSameTenant(resource.data) && isSameTenant(request.resource.data) && 
                            request.resource.data.keys().hasAll(['capThicknessIn', 'capAreaSqIn', 'capModulusMPa']) &&
                            request.resource.data.capThicknessIn is number &&
                            request.resource.data.capAreaSqIn is number &&
                            request.resource.data.capModulusMPa is number;
      allow delete: if hasTenant() && isSameTenant(resource.data);
    }
  }
}
