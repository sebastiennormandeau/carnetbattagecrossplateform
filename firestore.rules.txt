rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    function isAdmin() {
      return request.auth != null &&
             exists(/databases/$(database)/documents/admins/$(request.auth.uid)) &&
             get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.enabled == true;
    }

    match /users/{userId} {
      allow read: if request.auth != null; // All users must be able to read user tools settings if needed
      allow write: if isAdmin() || request.auth.uid == userId;
    }

    match /admins/{userId} {
      allow read: if request.auth != null;
      allow write: if isAdmin(); 
    }

    match /projects/{projectId} {
      allow read: if isAdmin() || 
                      resource.data.ownerUid == request.auth.uid ||
                      (resource.data.readUsers != null && request.auth.uid in resource.data.readUsers);
                      
      allow write: if isAdmin() || 
                       resource.data.ownerUid == request.auth.uid ||
                       (resource.data.writeUsers != null && request.auth.uid in resource.data.writeUsers);
      
      allow create: if request.auth != null;

      match /piles/{pileId} {
        allow read: if isAdmin() || 
                        get(/databases/$(database)/documents/projects/$(projectId)).data.ownerUid == request.auth.uid ||
                        (get(/databases/$(database)/documents/projects/$(projectId)).data.readUsers != null && request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.readUsers);
        allow write: if isAdmin() || 
                         get(/databases/$(database)/documents/projects/$(projectId)).data.ownerUid == request.auth.uid ||
                         (get(/databases/$(database)/documents/projects/$(projectId)).data.writeUsers != null && request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.writeUsers);
      }
      match /photos/{photoId} {
        allow read: if isAdmin() || 
                        get(/databases/$(database)/documents/projects/$(projectId)).data.ownerUid == request.auth.uid ||
                        (get(/databases/$(database)/documents/projects/$(projectId)).data.readUsers != null && request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.readUsers);
        allow write: if isAdmin() || 
                         get(/databases/$(database)/documents/projects/$(projectId)).data.ownerUid == request.auth.uid ||
                         (get(/databases/$(database)/documents/projects/$(projectId)).data.writeUsers != null && request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.writeUsers);
      }
      match /notes/{noteId} {
        allow read: if isAdmin() || 
                        get(/databases/$(database)/documents/projects/$(projectId)).data.ownerUid == request.auth.uid ||
                        (get(/databases/$(database)/documents/projects/$(projectId)).data.readUsers != null && request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.readUsers);
        allow write: if isAdmin() || 
                         get(/databases/$(database)/documents/projects/$(projectId)).data.ownerUid == request.auth.uid ||
                         (get(/databases/$(database)/documents/projects/$(projectId)).data.writeUsers != null && request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.writeUsers);
      }
      match /docs/{docId} {
        allow read: if isAdmin() || 
                        get(/databases/$(database)/documents/projects/$(projectId)).data.ownerUid == request.auth.uid ||
                        (get(/databases/$(database)/documents/projects/$(projectId)).data.readUsers != null && request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.readUsers);
        allow write: if isAdmin() || 
                         get(/databases/$(database)/documents/projects/$(projectId)).data.ownerUid == request.auth.uid ||
                         (get(/databases/$(database)/documents/projects/$(projectId)).data.writeUsers != null && request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.writeUsers);
      }
      match /hotspots/{hotspotId} {
        allow read: if isAdmin() || 
                        get(/databases/$(database)/documents/projects/$(projectId)).data.ownerUid == request.auth.uid ||
                        (get(/databases/$(database)/documents/projects/$(projectId)).data.readUsers != null && request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.readUsers);
        allow write: if isAdmin() || 
                         get(/databases/$(database)/documents/projects/$(projectId)).data.ownerUid == request.auth.uid ||
                         (get(/databases/$(database)/documents/projects/$(projectId)).data.writeUsers != null && request.auth.uid in get(/databases/$(database)/documents/projects/$(projectId)).data.writeUsers);
      }
    }

    match /map_points/{pointId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    match /settings/{docId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    match /shifts/{shiftId} {
      allow read: if isAdmin() || (request.auth != null && resource.data.userId == request.auth.uid);
      allow create: if request.auth != null && request.resource.data.userId == request.auth.uid;
      allow update: if request.auth != null && resource.data.userId == request.auth.uid;
    }

    match /hammers/{hammerId} {
      allow read, delete: if request.auth != null;
      allow create, update: if request.auth != null && 
                            request.resource.data.keys().hasAll(['capThicknessIn', 'capAreaSqIn', 'capModulusMPa']) &&
                            request.resource.data.capThicknessIn is number &&
                            request.resource.data.capAreaSqIn is number &&
                            request.resource.data.capModulusMPa is number;
    }
  }
}
